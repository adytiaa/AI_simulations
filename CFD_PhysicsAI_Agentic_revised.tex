\section{Motivation}
% ============================================================

\begin{frame}{CFD is accurate, but expensive}
\begin{columns}[T,onlytextwidth]
\column{0.52\textwidth}
\begin{block}{Why accelerate?}
\begin{itemize}\footnotesize
  \item \textbf{High cost per simulation:} high-fidelity, multi-physics CFD needs fine meshes and long runs.
  \item \textbf{The real cost is the sweep, not one run:} design studies vary pull rate, rotation rates, thermal boundary conditions, geometry\ldots{} each point is a new simulation.
  \item \textbf{Multi-physics coupling} (flow + heat + particles) adds model complexity.
  \item \textbf{Set-up expertise:} a valid case needs deep solver knowledge.
\end{itemize}
\end{block}
\vspace{0.1cm}
{\small\key{Goal:} keep the governing physics, cut the cost, and make set-up accessible.}

\column{0.45\textwidth}
\centering
\figbox{cz-process-schematic}{\linewidth}{3.8cm}{Silicon crystal-growth (CZ) process schematic: crucible, melt, crystal, rotation, heater}
\end{columns}
\end{frame}

\begin{columns}[T,onlytextwidth]
\column{0.56\textwidth}

\begin{itemize}\small

\end{block}
\begin{block}{Unknowns per point $(r,z)$}
\small $u_r,\;u_z,\;u_\theta$ (velocity), $p$ (pressure), $T$ (temperature): five coupled fields.
\end{block}

\section{Physics-informed learning}
% ============================================================

\begin{frame}{Physics-Informed Neural Networks (PINNs): the idea}

\begin{columns}[T,onlytextwidth]
\column{0.50\textwidth}
\begin{block}{Core idea}
\small A neural network $\hat u_\theta$ represents the solution. Training penalises both \textbf{data mismatch} and \textbf{physics violations}.
\end{block}
\begin{enumerate}\small
  \item Sample points in the domain and on its boundaries.
  \item Use automatic differentiation to evaluate PDE residuals.
  \item Optimise one objective combining PDE, BC and IC terms.
\end{enumerate}
\begin{exampleblock}{Why it is attractive}
\small Physics enters training directly, reducing reliance on labelled solution data.
\end{exampleblock}
\source{Raissi, Perdikaris \& Karniadakis (2019)}
\column{0.46\textwidth}
\centering
\figbox{pinn-concept}{\linewidth}{3.6cm}{PINN concept: coordinates $\rightarrow$ network $\rightarrow$ field; autodiff gives residuals; loss = PDE + BC + IC}
\end{columns}

\end{frame}

\begin{frame}{PINN methodology: physics becomes part of the objective}

\begin{columns}[T,onlytextwidth]
\column{0.54\textwidth}
\[
\boxed{
L_{\rm total}=L_{\rm PDE}+L_{\rm BC}+L_{\rm IC}+\lambda L_{\rm SAL}
}
\]
\begin{itemize}\small
  \item $L_{\rm PDE}$: governing-equation residual.
  \item $L_{\rm BC},L_{\rm IC}$: boundary and initial-condition mismatch.
  \item $\lambda$: relative weighting of an additional term.
\end{itemize}
\vspace{2mm}
\begin{alertblock}{The central issue}
\small The \textbf{weights matter}. A smaller weighted loss does not necessarily mean a more accurate solution.
\end{alertblock}
\column{0.43\textwidth}
\centering
\figbox{pinn-loss-diagram}{\linewidth}{4.0cm}{PINN loss composition / training workflow}
\end{columns}
\source{Raissi et al. (2019); Wang, Teng \& Perdikaris (2021)}

\end{frame}

\begin{frame}{Why PINN training is not automatically stable}

\begin{columns}[T,onlytextwidth]
\column{0.50\textwidth}
\begin{block}{Why optimisation can fail}
\begin{itemize}\small
  \item PDE and constraint terms can have very different gradient scales.
  \item Stiff or competing terms make the loss landscape difficult to optimise.
  \item A single aggregate loss can hide a badly satisfied constraint.
\end{itemize}
\end{block}
\column{0.46\textwidth}
\begin{exampleblock}{Implication for this study}
\small
Compare methods using:
\begin{enumerate}\small
  \item each loss component;
  \item an external solution error when a reference exists;
  \item physical validation where possible.
\end{enumerate}
\end{exampleblock}
\centering
\figbox{pinn-training-stability}{\linewidth}{2.4cm}{Illustration of competing loss terms / gradient imbalance}
\end{columns}
\source{Wang, Teng \& Perdikaris (2021); Krishnapriyan et al. (2021)}

\end{frame}

\begin{frame}{From PINNs to neural operators}

\begin{columns}[T,onlytextwidth]
\column{0.50\textwidth}
\begin{block}{From one solution to a solution family}
\small
A PINN typically solves one PDE instance. A \textbf{neural operator} learns a map
\[
\text{inputs: coefficients / geometry / conditions}
\;\longrightarrow\;
\text{solution field}.
\]
\end{block}
\begin{itemize}\small
  \item One model can serve many parameter settings.
  \item Physics-informed operators add PDE constraints during training.
  \item This is better aligned with repeated-query CFD workflows.
\end{itemize}
\column{0.46\textwidth}
\centering
\figbox{pino-architecture}{\linewidth}{5.0cm}{PINO / physics-informed operator: input function $\rightarrow$ operator network $\rightarrow$ solution function, with physics loss}
\end{columns}
\source{Kovachki et al. (2023); PI-GANO, Zhong \& Meidani (2024)}

\end{frame}

\begin{frame}{Geometry-aware operators: the next generalisation step}

\begin{columns}[T,onlytextwidth]
\column{0.50\textwidth}
\begin{block}{Why geometry matters}
\small Real CFD design spaces change both \textbf{parameters} and \textbf{shape}. Geometry-aware operators encode the domain rather than assuming one fixed grid.
\end{block}
\begin{itemize}\small
  \item \textbf{PI-GANO}: physics-informed + geometry-aware operator learning.
  \item \textbf{GAOT / MAGNO}: attention / graph-based operator learning on irregular geometry.
\end{itemize}
\begin{exampleblock}{CFD implication}
\small The target is not one surrogate for one case, but a reusable solution map across operating conditions and geometries.
\end{exampleblock}
\column{0.46\textwidth}
\centering
\figbox{geometry-aware-operators}{\linewidth}{4.2cm}{PI-GANO / GAOT architecture overview}
\end{columns}
\source{Zhong \& Meidani (2024), arXiv:2408.01600; GAOT/MAGNO (2025)}

\end{frame}

\begin{frame}{Comparing the classical approaches}

\small
\renewcommand{\arraystretch}{1.15}
\begin{tabularx}{\textwidth}{>{\bfseries}l L L L}
\toprule
Approach & What it learns & Main advantage & Main caveat \\
\midrule
PINN & One PDE solution & Physics constraints & Training stability / scaling \\
PINO & Solution operator & Reusable across parameters & Needs representative coverage \\
PI-GANO & Geometry-aware operator & Parameter + geometry generalisation & Geometry encoding / validation \\
GAOT & Graph / attention operator & Irregular-domain modelling & Cost and validation \\
\bottomrule
\end{tabularx}
\vspace{0.2cm}
\centering
\figbox{approach-comparison}{0.7\linewidth}{1.9cm}{Optional: comparison figure / benchmark chart}
\vfill
\raggedright\source{Kovachki et al. (2023); Zhong \& Meidani (2024); GAOT/MAGNO (2025)}

\end{frame}

\section{Case study: crystal growth}
% ============================================================

\begin{frame}{Case study overview: two experiments, two questions}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{A. Adaptive PINN training}
\small
\textbf{Question:} when loss weights adapt during training, is a lower PDE loss really a better solution?\\[3pt]
Problems: (1) 1D heat equation with an analytical answer; (2) simplified Czochralski thermal-fluid PINN.\\[3pt]
Compared: Baseline PINN, GNPINN, AgenticPINN.
\end{block}
\column{0.48\textwidth}
\begin{block}{B. CFD surrogates (Gaussian process)}
\small
\textbf{Question:} how well does a surrogate predict a \emph{whole unseen operating condition}, field by field?\\[3pt]
Data: ANSYS Fluent reference CFD, three one-parameter sweeps.\\[3pt]
Metric: NRMSE per output field.
\end{block}
\end{columns}
\vspace{0.2cm}
\begin{alertblock}{Shared message}
\small Aggregate numbers hide where a model fails. Report component-level losses and field-wise errors.
\end{alertblock}
\vspace{0.1cm}
\smallnote{Work with N.\ Chapagain, R.\ Raj, B.\ Kurisinkal Shine (NeurIPS 2026 Sim2Science workshop).}
\end{frame}

\begin{frame}{Case study: CFD reference data for Czochralski growth}
\begin{columns}[T,onlytextwidth]
\column{0.54\textwidth}
\begin{block}{Reference CFD}
\begin{itemize}\footnotesize
  \item ANSYS Fluent 2026 R1; \textbf{steady, laminar, 2D axisymmetric} melt.
  \item Incompressible Navier--Stokes + energy, Boussinesq buoyancy, azimuthal velocity from rotation.
  \item Rotating walls: $u_\theta=\omega r$; free surface non-penetrating and slip; prescribed thermal conditions at free surface and crucible bottom.
  \item Mesh-independence study $\Rightarrow$ common grid of 8181 nodes.
  \item Rotational Reynolds numbers are large, so these are \emph{controlled reference cases}, not fully resolved industrial melt flow.
\end{itemize}
\end{block}
\column{0.43\textwidth}
\centering
\figbox{cz-cfd-reference}{\linewidth}{4.6cm}{Reference CFD fields (temperature, velocity components) in $(r,z)$}
\end{columns}
\end{frame}

\begin{frame}{Three parameter sweeps and the case-wise holdout}
\begin{columns}[T,onlytextwidth]
\column{0.52\textwidth}
\begin{block}{One-parameter sweeps}
\small
\begin{tabular}{@{}lll@{}}
\toprule
Parameter & Range & Held-out case \\
\midrule

\end{tabular}
\end{block}
\begin{block}{Surrogate learns}
\small
\[(r,z,s)\;\mapsto\;(u_r,u_z,u_\theta,p,T)\]
\small $s$ is the active process parameter of the sweep.
\end{block}
\column{0.45\textwidth}
\begin{alertblock}{Why a case-wise holdout?}
\small
One \emph{entire operating condition} is withheld for testing. This measures generalisation to an unseen process setting, not interpolation between randomly split spatial points of cases already seen.
\end{alertblock}
\centering
\figbox{holdout-schematic}{\linewidth}{2.6cm}{Sweep values on a line; one value held out for test}
\end{columns}
\end{frame}

\begin{frame}{Benchmark 1: the 1D heat equation (analytical reference)}

\begin{columns}[T,onlytextwidth]
\column{0.55\textwidth}
\[
\frac{\partial T}{\partial t}=\alpha\frac{\partial^2T}{\partial x^2},
\quad T(0,t)=T(1,t)=0,\quad T(x,0)=\sin(\pi x)
\]
\eqbox{$T_{\rm exact}(x,t)=\sin(\pi x)e^{-\alpha\pi^2t}$}
\begin{itemize}\small
  \item $\alpha=1$ in the benchmark.
  \item Exact solution gives a direct \textbf{field-accuracy} metric.
  \item This is the control experiment for interpreting PINN losses.
\end{itemize}
\begin{alertblock}{Important}
\small The crystal-growth case has no analytical reference; its losses are constraint diagnostics, not proof of accuracy.
\end{alertblock}
\column{0.42\textwidth}
\centering
\figbox{heat-equation-exact}{\linewidth}{4.8cm}{Analytical heat-equation solution $T(x,t)$ over $[0,1]^2$}
\end{columns}

\end{frame}

\begin{frame}{Adaptive PINN training: what do we compare?}

\begin{columns}[T,onlytextwidth]
\column{0.56\textwidth}
\begin{block}{Matched comparison}
\begin{itemize}\small
  \item \textbf{Baseline PINN}: fixed unit weights.
  \item \textbf{GNPINN}: normalises PDE and BC/IC gradients.
  \item \textbf{AgenticPINN}: rule-based controller adapts the weights.
\end{itemize}
\end{block}
\[
\mathcal L=\lambda_{\rm PDE}\mathcal L_{\rm PDE}
+\lambda_{\rm BC/IC}\mathcal L_{\rm BC/IC}
\]
\begin{exampleblock}{Fair comparison}
\small Same architecture, optimiser, sampling budget, seed and learning-rate schedule within each experiment.
\end{exampleblock}
\column{0.40\textwidth}
\centering
\figbox{training-strategies-overview}{\linewidth}{4.0cm}{Three trainers: fixed weights / normalised gradients / adaptive controller}
\end{columns}
\vspace{1mm}\smallnote{AgenticPINN here is a rule-based controller; it is distinct from the LLM-based AutoFOAM agent.}

\end{frame}

\begin{frame}{The three strategies in detail}

\begin{columns}[T,onlytextwidth]
\column{0.32\textwidth}
\begin{block}{Baseline PINN}
\footnotesize Fixed unit weights; minimise the combined PDE + BC/IC loss.
\end{block}
\column{0.32\textwidth}
\begin{block}{GNPINN}
\footnotesize Compute PDE and BC/IC gradients separately, normalise them, then combine.
\end{block}
\column{0.32\textwidth}
\begin{block}{AgenticPINN}
\footnotesize Monitor gradient norms and update the two loss weights with bounded, smoothed rules.
\end{block}
\end{columns}
\vspace{0.2cm}
\centering
\begin{exampleblock}{What changes?}
\small The network and physics stay fixed; only the \textbf{training objective / gradient balancing} changes.
\end{exampleblock}

\end{frame}

\begin{frame}{AgenticPINN: the controller rule}

\begin{columns}[T,onlytextwidth]
\column{0.55\textwidth}
\footnotesize
With smoothed gradient norms $\bar g_{\rm PDE}$ and $\bar g_{\rm BC/IC}$:
\[
a=\operatorname{clip}\!\left(
\sqrt{\frac{\bar g_{\rm PDE}+10^{-12}}
{\bar g_{\rm BC/IC}+10^{-12}}},\,0.8,\,1.25\right)
\]
\[
\lambda_{\rm PDE}\leftarrow\operatorname{clip}(\lambda_{\rm PDE}/a),\qquad
\lambda_{\rm BC/IC}\leftarrow\operatorname{clip}(\lambda_{\rm BC/IC}a).
\]
\begin{itemize}\footnotesize
  \item Weight changes are bounded; weights remain in $[10^{-2},10^2]$.
  \item A learning-rate reduction is triggered when progress stalls.
\end{itemize}
\column{0.42\textwidth}
\begin{exampleblock}{Intuition}
\small Balance the \textbf{gradient influence} of PDE and constraints.
\end{exampleblock}
\centering
\figbox{controller-flow}{\linewidth}{2.6cm}{Controller loop: measure gradient norms $\rightarrow$ update weights $\rightarrow$ train}
\end{columns}

\end{frame}

\begin{frame}{Why the weighted total is not a fair score}

\begin{columns}[T,onlytextwidth]
\column{0.52\textwidth}
\begin{block}{Three different questions}
\small
\begin{enumerate}
  \item Did the optimiser reduce the chosen objective?
  \item Are all physical constraints satisfied?
  \item Is the predicted field accurate?
\end{enumerate}
\end{block}
\[
\mathcal L=\sum_k\lambda_k\mathcal L_k
\]
\small If $\lambda_k$ changes between methods, \textbf{weighted totals are not directly comparable}.
\column{0.44\textwidth}

\begin{frame}{CFD surrogate: Gaussian-process set-up}
\begin{columns}[T,onlytextwidth]
\column{0.55\textwidth}
\begin{block}{Model}
\footnotesize
\begin{itemize}
  \item Multi-output GPR for $(r,z,s)\mapsto(u_r,u_z,u_\theta,p,T)$, inputs and targets scaled.
  \item Kernel: constant $\times$ anisotropic RBF (one length scale per input) $+$ white noise ($10^{-4}$); extra diagonal term $10^{-6}$.
  \item Kernel hyperparameters are fixed, not optimised during fitting.
  \item Returns predictive mean and standard deviation, mapped back to physical units.
\end{itemize}
\end{block}
\column{0.42\textwidth}
\begin{block}{Metric}
\footnotesize
\[
\mathrm{NRMSE}=\frac{\sqrt{\frac1N\sum_i(\hat y_i-y_i)^2}}{\sigma_y}
\]
$\sigma_y$: standard deviation of that field in the holdout case. Normalisation lets us compare fields with different units and scales.
\end{block}
\begin{exampleblock}{Uncertainty diagnostic}
\footnotesize Pearson correlation between $|\hat y-y|$ and predictive std; an error-awareness check, not a formal calibration.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Case study: what we learn}
\begin{columns}[T,onlytextwidth]
\column{0.32\textwidth}
\begin{block}{1. Objective $\ne$ accuracy}
\small Adaptively weighted objectives are optimisation quantities, not common cross-method accuracy metrics.
\end{block}

\small Unweighted PDE, BC and IC losses, plus external solution error where available.

\begin{frame}{Key design choices}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Constrained parameter extraction}
\begin{itemize}\small
  \item \textbf{vLLM} for high-throughput inference.
  \item \textbf{xgrammar} JSON schema enforces the \texttt{CFDParams} structure, so output is always parseable.
  \item Output: geometry, $Re$, fluid, solver, turbulence model.
  \item Regex check of $Re=UL/\nu$ consistency between stated and derived values.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Deterministic solver routing}
\begin{itemize}\small
  \item Decouples solver choice from the LLM.
  \item Physics flags: transient, compressible, thermal, multiphase.
  \item Routes to one of \textbf{7 approved solvers}.
  \item Avoids LLM bias, e.g.\ \texttt{simpleFoam} for transient vortex shedding.
\end{itemize}
\end{block}
\end{columns}
\vspace{0.1cm}
\begin{block}{Design principle}
\small Use the LLM for language understanding; use rules and code wherever a wrong answer would crash a run or give a wrong physics set-up.
\end{block}

\column{0.32\textwidth}
\begin{exampleblock}{RAG-augmented retry}
\footnotesize Chroma vector DB of verified OpenFOAM tutorial configs; relevant dictionary snippets are injected into the retry prompt, grounding the fix in known-good files.
\end{exampleblock}
\column{0.32\textwidth}
\begin{exampleblock}{Surgical dictionary patching}
\footnotesize On localised \texttt{FOAM FATAL} errors only the offending dictionary is patched, skipping costly \texttt{gmsh} remeshing.
\end{exampleblock}
\column{0.32\textwidth}
\begin{exampleblock}{Prompt-diversity paraphrasing}
\footnotesize Varies the phrasing of training prompts so the model does not overfit one pattern.
\end{exampleblock}

\begin{frame}{Limitations and validation gaps}

\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{alertblock}{AutoFOAM}
\small
\begin{itemize}
  \item 13 parameterised geometries; no raw CAD pipeline.
  \item RANS-focused; broader physics remains to be tested.
  \item Reward favours execution / convergence and can miss physical error.
\end{itemize}
\end{alertblock}
\column{0.48\textwidth}
\begin{alertblock}{Physics-informed surrogates}
\small
\begin{itemize}
  \item Small benchmark set and controlled initialisation.
  \item Crystal-growth case lacks a ground-truth field.
  \item Field difficulty is uneven: velocity can be much harder than temperature.
\end{itemize}
\end{alertblock}
\end{columns}
\vspace{0.15cm}
\begin{exampleblock}{Priority}
\small Add independent benchmark checks and field/component-level metrics before claiming general CFD accuracy.
\end{exampleblock}

\begin{frame}{Key takeaways}

\begin{enumerate}\setlength{\itemsep}{4pt}\small
  \item \key{CFD is the bottleneck:} high-fidelity simulation is expensive and expertise-heavy.
  \item \key{Physics helps, but metrics matter:} PINNs can reduce PDE residuals while producing worse fields.
  \item \key{Operators scale the idea:} learn reusable maps across parameters and, increasingly, geometry.
  \item \key{Surrogate error is field-specific:} temperature can be accurate while velocity remains difficult.
  \item \key{Agents remove set-up friction:} AutoFOAM automates case generation, execution and correction.
  \item \key{Validation is the common layer:} neither a low loss nor a converged CFD run is sufficient on its own.
\end{enumerate}

\large\textcolor{midgray}{$\longrightarrow$ accelerated CFD with explicit validation}
